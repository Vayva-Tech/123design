## ffprobe Batch Command

```bash
for f in "Images & Videos/VIDEO/"*.mp4; do
  echo "=== $(basename "$f") ==="
  ffprobe -v quiet -print_format json -show_streams "$f" | \
    python3 -c "import sys,json; d=json.load(sys.stdin);
s=[x for x in d["streams"] if x["codec_type"]=="video"];
print(f"  {s[0]["width"]}x{s[0]["height"]} {s[0].get("duration","?")}s {s[0].get("codec_name","?")}") if s else print("  no video stream")"
done
```
